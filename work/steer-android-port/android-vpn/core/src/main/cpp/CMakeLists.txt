# Run the upstream assertions on Android. Only their /tmp paths are adapted;
# source files and checks in the upstream tree remain unchanged.
function(steer_test NAME)
    file(READ "${STEER}/tests/${NAME}.c" BODY)
    string(REPLACE "/tmp/" "/data/local/tmp/" BODY "${BODY}")
    string(REPLACE "../src/" "${STEER}/src/" BODY "${BODY}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${NAME}.c" "${BODY}")
    add_executable(${NAME} EXCLUDE_FROM_ALL "${CMAKE_CURRENT_BINARY_DIR}/${NAME}.c" ${ARGN})
    target_include_directories(${NAME} PRIVATE ${INCLUDES} "${STEER}/tests")
    target_compile_options(${NAME} PRIVATE -UNDEBUG -ffunction-sections -fdata-sections)
    target_link_options(${NAME} PRIVATE -Wl,--gc-sections)
endfunction()
